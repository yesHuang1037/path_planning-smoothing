cmake_minimum_required(VERSION 3.16)
project(tdt_path_planning)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core highgui imgproc)
find_package(OsqpEigen REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(map_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)

option(TDT_SANITIZE "Enable address/undefined sanitizers for this package" OFF)
if(TDT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(tdt_planning_core
  src/grid.cpp src/search.cpp src/optimizer.cpp src/snapshot_io.cpp
  vendor/dynamicvoronoi.cpp vendor/bucketedqueue.cpp)
target_include_directories(tdt_planning_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include> PRIVATE vendor ${OpenCV_INCLUDE_DIRS})
target_link_libraries(tdt_planning_core PUBLIC Eigen3::Eigen OsqpEigen::OsqpEigen PRIVATE ${OpenCV_LIBS})
target_compile_options(tdt_planning_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(offline_planner src/offline.cpp)
target_link_libraries(offline_planner tdt_planning_core)
add_executable(snapshot_planner src/node.cpp)
target_link_libraries(snapshot_planner tdt_planning_core)
ament_target_dependencies(snapshot_planner rclcpp nav_msgs map_msgs geometry_msgs std_msgs visualization_msgs tf2_ros tf2_geometry_msgs)

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(core_test test/core_test.cpp)
  target_link_libraries(core_test tdt_planning_core)
endif()
install(TARGETS tdt_planning_core offline_planner snapshot_planner
  ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY cfg launch fixtures DESTINATION share/${PROJECT_NAME})
install(FILES SOURCES.md DESTINATION share/${PROJECT_NAME})
ament_export_include_directories(include)
ament_export_libraries(tdt_planning_core)
ament_export_dependencies(Eigen3 OsqpEigen)
ament_package()
